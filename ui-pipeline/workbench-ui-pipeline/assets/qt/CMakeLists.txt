cmake_minimum_required(VERSION 3.24)
project(WorkbenchSample LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
  add_compile_options(/utf-8 /EHsc)
endif()
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(workbench_sample main.cpp)
target_link_libraries(workbench_sample PRIVATE Qt6::Widgets)
target_compile_definitions(workbench_sample PRIVATE WORKBENCH_THEME_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
