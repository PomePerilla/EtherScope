cmake_minimum_required(VERSION 3.24)
project(EtherScope VERSION 1.9.0 LANGUAGES CXX)
configure_file(src/version.h.in generated/version.h @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive- /EHsc)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
include(CTest)
add_library(etherscope_core src/core/packet.cpp src/core/filter.cpp src/core/pcap_file.cpp src/core/session.cpp src/core/tcp_conversations.cpp src/core/tcp_stream.cpp)
target_include_directories(etherscope_core PUBLIC src)
option(ETHERSCOPE_BUILD_GUI "Build the Qt desktop application" ON)
option(ETHERSCOPE_BUILD_FUZZERS "Build parser and TCP stream fuzz targets" OFF)
if(ETHERSCOPE_BUILD_FUZZERS)
  if(NOT MSVC AND (NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR WIN32))
    message(FATAL_ERROR "Fuzz targets require MSVC 2022 or Linux Clang")
  endif()
  add_executable(packet_fuzzer tests/packet_fuzzer.cpp src/core/packet.cpp)
  add_executable(stream_fuzzer tests/stream_fuzzer.cpp src/core/tcp_stream.cpp src/core/tcp_conversations.cpp src/core/session.cpp src/core/packet.cpp)
  foreach(fuzzer packet_fuzzer stream_fuzzer)
    target_include_directories(${fuzzer} PRIVATE src)
    if(MSVC)
      target_compile_options(${fuzzer} PRIVATE /fsanitize=address /fsanitize=fuzzer /Z7)
      target_link_options(${fuzzer} PRIVATE /INCREMENTAL:NO /DEBUG)
    else()
      target_compile_options(${fuzzer} PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
      target_link_options(${fuzzer} PRIVATE -fsanitize=fuzzer,address,undefined)
    endif()
  endforeach()
endif()
if(ETHERSCOPE_BUILD_GUI)
  find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets)
  if(BUILD_TESTING)
    find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Test)
  endif()
  add_library(etherscope_desktop
    src/capture/capture.cpp
    src/capture/preparation.cpp
    src/capture/export.cpp
    src/capture/details.cpp
    src/capture/stream.cpp
    src/ui/main_window.cpp
    src/ui/language.cpp
    src/ui/main_window_ui.cpp
    src/ui/main_window_theme.cpp
    src/ui/main_window_filter.cpp
    src/ui/filter_chip.cpp
    src/ui/main_window_settings.cpp
    src/ui/main_window_inspector.cpp
    src/ui/main_window_tcp.cpp
    src/ui/tcp_dialog.cpp
    src/ui/stream_pane.cpp
    src/ui/packet_model.cpp
    src/ui/traffic_chart.cpp)
  set_target_properties(etherscope_desktop PROPERTIES AUTOMOC ON)
  target_include_directories(etherscope_desktop PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(etherscope_desktop PUBLIC etherscope_core Qt6::Widgets)
  qt_add_resources(etherscope_desktop desktop_icons PREFIX "/icons" BASE "src/ui/icons" FILES src/ui/icons/chevron-down.xpm)
  qt_add_resources(etherscope_desktop localization PREFIX "/i18n" BASE "src/ui/i18n" FILES src/ui/i18n/messages.json)
  if(WIN32)
    find_path(NPCAP_INCLUDE_DIR pcap/pcap.h HINTS "${NPCAP_SDK_DIR}/Include" "${CMAKE_SOURCE_DIR}/.tools/npcap-sdk/Include" REQUIRED)
    target_include_directories(etherscope_desktop PRIVATE "${NPCAP_INCLUDE_DIR}")
    target_link_libraries(etherscope_desktop PRIVATE ws2_32)
  endif()
  qt_add_executable(EtherScope WIN32 src/main.cpp)
  if(WIN32)
    configure_file(src/version.rc.in generated/version.rc @ONLY)
    target_sources(EtherScope PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/version.rc")
  endif()
  target_link_libraries(EtherScope PRIVATE etherscope_desktop)
  install(TARGETS EtherScope RUNTIME DESTINATION .)
  if(BUILD_TESTING)
    qt_add_executable(ui_tests tests/ui_tests.cpp tests/fixtures.cpp)
    set_target_properties(ui_tests PROPERTIES AUTOMOC ON)
    target_link_libraries(ui_tests PRIVATE etherscope_desktop Qt6::Test)
    add_test(NAME desktop_interactions COMMAND ui_tests -platform offscreen -o ui-tests.xml,junitxml)
    qt_add_executable(ui_benchmark tests/ui_benchmark.cpp tests/fixtures.cpp)
    target_link_libraries(ui_benchmark PRIVATE etherscope_desktop)
    qt_add_executable(capture_probe tests/capture_probe.cpp)
    target_link_libraries(capture_probe PRIVATE etherscope_desktop)
    if(WIN32)
      target_link_libraries(ui_benchmark PRIVATE psapi)
      add_executable(bpf_tests tests/bpf_tests.cpp tests/fixtures.cpp)
      target_include_directories(bpf_tests PRIVATE "${NPCAP_INCLUDE_DIR}")
      target_link_libraries(bpf_tests PRIVATE etherscope_core)
    endif()
  endif()
endif()
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp tests/fixtures.cpp)
  target_link_libraries(core_tests PRIVATE etherscope_core)
  add_test(NAME protocol_filter_pcap_session COMMAND core_tests)
  add_executable(tcp_tests tests/tcp_tests.cpp tests/fixtures.cpp)
  target_link_libraries(tcp_tests PRIVATE etherscope_core)
  add_test(NAME tcp_conversations COMMAND tcp_tests)
  add_executable(stream_tests tests/stream_tests.cpp tests/fixtures.cpp)
  target_link_libraries(stream_tests PRIVATE etherscope_core)
  add_test(NAME tcp_stream_reassembly COMMAND stream_tests)
  add_executable(etherscope_benchmark tests/benchmark.cpp tests/fixtures.cpp)
  target_link_libraries(etherscope_benchmark PRIVATE etherscope_core)
  add_executable(make_samples tests/make_samples.cpp tests/fixtures.cpp)
  target_link_libraries(make_samples PRIVATE etherscope_core)
  add_executable(make_fuzz_corpus tests/make_fuzz_corpus.cpp tests/fixtures.cpp)
  target_link_libraries(make_fuzz_corpus PRIVATE etherscope_core)
  add_executable(inspect_pcap tests/inspect_pcap.cpp)
  target_link_libraries(inspect_pcap PRIVATE etherscope_core)
endif()
