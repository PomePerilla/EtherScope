cmake_minimum_required(VERSION 3.24)
project(EtherScope VERSION 1.10.0 LANGUAGES CXX)
configure_file(src/version.h.in generated/version.h @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive- /EHsc)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
include(CTest)
add_library(etherscope_core src/core/packet.cpp src/core/filter.cpp src/core/pcap_file.cpp src/core/capture_file.cpp src/core/session.cpp src/core/tcp_conversations.cpp src/core/tcp_stream.cpp src/core/application.cpp)
target_sources(etherscope_core PRIVATE src/core/http.cpp)
target_sources(etherscope_core PRIVATE src/core/display_filter.cpp)
target_sources(etherscope_core PRIVATE src/core/application_analyzer.cpp)
target_sources(etherscope_core PRIVATE src/core/application_index.cpp)
list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/.tools/dependencies/install")
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(OpenSSL 3.5.9 EXACT REQUIRED COMPONENTS Crypto)
find_package(nghttp2 1.70.0 EXACT CONFIG REQUIRED)
target_sources(etherscope_core PRIVATE src/core/tls_crypto.cpp src/core/tls.cpp src/core/http2.cpp)
target_link_libraries(etherscope_core PUBLIC OpenSSL::Crypto nghttp2::nghttp2_static)
target_compile_definitions(etherscope_core PUBLIC NGHTTP2_NO_SSIZE_T)
target_include_directories(etherscope_core PUBLIC src)
option(ETHERSCOPE_BUILD_GUI "Build the Qt desktop application" ON)
option(ETHERSCOPE_BUILD_FUZZERS "Build parser and TCP stream fuzz targets" OFF)
if(ETHERSCOPE_BUILD_FUZZERS)
  if(NOT MSVC AND (NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR WIN32))
    message(FATAL_ERROR "Fuzz targets require MSVC 2022 or Linux Clang")
  endif()
  add_executable(packet_fuzzer tests/packet_fuzzer.cpp src/core/packet.cpp)
  add_executable(stream_fuzzer tests/stream_fuzzer.cpp src/core/tcp_stream.cpp src/core/tcp_conversations.cpp src/core/session.cpp src/core/packet.cpp)
  get_target_property(core_sources etherscope_core SOURCES)
  add_library(etherscope_fuzz_core STATIC ${core_sources})
  target_include_directories(etherscope_fuzz_core PUBLIC src)
  target_link_libraries(etherscope_fuzz_core PUBLIC OpenSSL::Crypto nghttp2::nghttp2_static)
  target_compile_definitions(etherscope_fuzz_core PUBLIC NGHTTP2_NO_SSIZE_T)
  add_executable(application_fuzzer tests/application_fuzzer.cpp)
  add_executable(file_fuzzer tests/file_fuzzer.cpp)
  target_link_libraries(application_fuzzer PRIVATE etherscope_fuzz_core)
  target_link_libraries(file_fuzzer PRIVATE etherscope_fuzz_core)
  if(MSVC)
    target_compile_options(etherscope_fuzz_core PRIVATE /fsanitize=address /fsanitize=fuzzer /Z7)
  else()
    target_compile_options(etherscope_fuzz_core PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
  endif()
  foreach(fuzzer packet_fuzzer stream_fuzzer application_fuzzer file_fuzzer)
    target_include_directories(${fuzzer} PRIVATE src)
    if(MSVC)
      target_compile_options(${fuzzer} PRIVATE /fsanitize=address /fsanitize=fuzzer /Z7)
      target_link_options(${fuzzer} PRIVATE /INCREMENTAL:NO /DEBUG)
    else()
      target_compile_options(${fuzzer} PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
      target_link_options(${fuzzer} PRIVATE -fsanitize=fuzzer,address,undefined)
    endif()
  endforeach()
endif()

    target_sources(EtherScope PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/version.rc")
  endif()
  target_link_libraries(EtherScope PRIVATE etherscope_desktop)
  install(TARGETS EtherScope RUNTIME DESTINATION .)
